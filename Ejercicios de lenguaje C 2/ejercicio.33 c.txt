#include <stdio.h>

// Función para buscar la primera ocurrencia de un valor en un vector
// Retorna el índice (0 a tamano-1) o -1 si no lo encuentra
int buscarValor(int vector[], int tamano, int valorBuscado) {
    for (int i = 0; i < tamano; i++) {
        if (vector[i] == valorBuscado) {
            return i; // Retorna la posición de la primera ocurrencia
        }
    }
    return -1; // El elemento no está en el vector
}

int main() {
    int N, valor, posicion;

    printf("Ingrese el tamano del vector: ");
    if (scanf("%d", &N) != 1 || N <= 0) {
        printf("Tamano invalido.\n");
        return 1;
    }

    int miVector[N];

    // Llenado del vector
    for (int i = 0; i < N; i++) {
        printf("Ingrese el elemento [%d]: ", i);
        scanf("%d", &miVector[i]);
    }

    printf("\nIngrese el valor a buscar: ");
    scanf("%d", &valor);

    // Llamada a la función
    posicion = buscarValor(miVector, N, valor);

    // Resultado
    if (posicion != -1) {
        printf("El valor %d se encuentra en la posicion (indice): %d\n", valor, posicion);
    } else {
        printf("El valor %d no esta en el vector. Resultado: %d\n", valor, posicion);
    }

    return 0;
}
