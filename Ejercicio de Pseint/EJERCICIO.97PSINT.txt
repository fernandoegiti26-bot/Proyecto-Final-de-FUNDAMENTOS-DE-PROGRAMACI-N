Algoritmo PiedraPapelTijeras
	Definir opcionUsuario, opcionCPU Como Entero
	
	// Inicializar el generador de números aleatorios
	// Reinicia la semilla aleatoria
	
	Escribir "=== PIEDRA, PAPEL O TIJERAS ==="
	Escribir "1. Piedra"
	Escribir "2. Papel"
	Escribir "3. Tijeras"
	Escribir "Elige una opcion (1-3): "
	Leer opcionUsuario
	
	// Validar la entrada del usuario
	Si opcionUsuario < 1 O opcionUsuario > 3 Entonces
		Escribir "Opcion invalida. Debes elegir 1, 2 o 3."
	Sino
		// Generar la opcion de la CPU (1: Piedra, 2: Papel, 3: Tijeras)
		// Aleatorio(1, 3) genera un número entero entre 1 y 3 inclusive
		opcionCPU <- Aleatorio(1, 3)
		
		// Mostrar la elección del usuario
		Escribir "Tu elegiste: " Sin Saltar
		Segun opcionUsuario Hacer
			1: Escribir "Piedra"
			2: Escribir "Papel"
			3: Escribir "Tijeras"
		FinSegun
		
		// Mostrar la elección de la CPU
		Escribir "La CPU eligio: " Sin Saltar
		Segun opcionCPU Hacer
			1: Escribir "Piedra"
			2: Escribir "Papel"
			3: Escribir "Tijeras"
		FinSegun
		
		Escribir "" // Línea en blanco para separar
		
		// Determinar el ganador
		Si opcionUsuario = opcionCPU Entonces
			Escribir "¡Es un EMPATE!"
		Sino
			Si (opcionUsuario = 1 Y opcionCPU = 3) O (opcionUsuario = 2 Y opcionCPU = 1) O (opcionUsuario = 3 Y opcionCPU = 2) Entonces
				Escribir "¡Felicidades! ¡HAS GANADO!"
			Sino
				Escribir "¡Lo siento! ¡HAS PERDIDO!"
			FinSi
		FinSi
	FinSi
FinAlgoritmo
